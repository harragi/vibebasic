Table Definitions (SQL DDL)

Complete CREATE TABLE statements for all tables

institutions Master Table 
CREATE TABLE institutions (
    institution_id SERIAL PRIMARY KEY,
    ipeds_id INTEGER UNIQUE NOT NULL,
    opeid VARCHAR(10),
    ope6_id VARCHAR(6),
    name TEXT NOT NULL,
    alias TEXT,
    institution_control VARCHAR(50) NOT NULL,  -- Public, Private nonprofit, Private for-profit
    sector_code INTEGER,
    level_code INTEGER,
    highest_degree VARCHAR(50),
    degree_granting BOOLEAN,
    is_active BOOLEAN DEFAULT TRUE,
    status VARCHAR(20),  -- Active, Inactive, Closed
    death_year INTEGER,
    closed_date DATE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);
institution_identifiers Domain Table 
CREATE TABLE institution_identifiers (
    id SERIAL PRIMARY KEY,
    institution_id INTEGER NOT NULL REFERENCES institutions(institution_id),
    identifier_type VARCHAR(50) NOT NULL,  -- ipeds, opeid, ope6, ein, uei, duns
    identifier_value VARCHAR(50) NOT NULL,
    is_primary BOOLEAN DEFAULT FALSE,
    valid_from DATE,
    valid_to DATE,
    UNIQUE(institution_id, identifier_type, identifier_value)
);

institution_locations Domain Table 
CREATE TABLE institution_locations (
    id SERIAL PRIMARY KEY,
    institution_id INTEGER NOT NULL REFERENCES institutions(institution_id),

    -- Address
    address TEXT,
    city VARCHAR(100) NOT NULL,
    state_code CHAR(2) NOT NULL,
    state_fips INTEGER,
    zip_code VARCHAR(10),
    county_name VARCHAR(100),
    county_fips VARCHAR(5),
    country VARCHAR(100) DEFAULT 'United States',

    -- Coordinates
    latitude NUMERIC(9,6),
    longitude NUMERIC(10,6),

    -- NCES Locale (11-43 codes)
    locale_code INTEGER,
    locale_type VARCHAR(20),  -- city, suburb, town, rural
    locale_size VARCHAR(20),  -- large, midsize, small, fringe, distant, remote

    -- Statistical Areas
    cbsa_code VARCHAR(5),
    cbsa_name VARCHAR(200),
    cbsa_type VARCHAR(20),  -- metropolitan, micropolitan
    csa_code VARCHAR(5),
    csa_name VARCHAR(200),

    -- Political
    congressional_district VARCHAR(10),
    region_code INTEGER,
    region_name VARCHAR(50),

    -- Attribution
    data_year INTEGER NOT NULL,
    source VARCHAR(50) NOT NULL DEFAULT 'ipeds_hd',
    confidence VARCHAR(20) DEFAULT 'high',
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),

    UNIQUE(institution_id, data_year)
);

Data Attribution Pattern

Every table includes standard attribution columns

Standard Attribution Columns All Tables 
-- Every domain table includes these columns:
data_year INTEGER NOT NULL,        -- Year the data represents
source VARCHAR(50) NOT NULL,    -- Source code (ipeds_hd, cds, scorecard)
source_year INTEGER,              -- Year the source was published
confidence VARCHAR(20) DEFAULT 'high',  -- high, medium, low, estimated
created_at TIMESTAMPTZ DEFAULT NOW(),
updated_at TIMESTAMPTZ DEFAULT NOW()

-- Unique constraint pattern (allows multiple sources per year):
UNIQUE(institution_id, data_year, source)
